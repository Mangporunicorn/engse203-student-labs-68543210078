-- ⑤ คำร้องพร้อมชื่อผู้แจ้ง  ← ต้องใช้ JOIN เพราะชื่ออยู่คนละตาราง
SELECT
  r.id,
  u.name AS requesterName,
  r.request_type,
  r.location,
  r.status
FROM requests AS r
JOIN users AS u
  ON r.requester_id = u.id
ORDER BY r.id;

-- ⑥ คำร้องเฉพาะของภาควิชาหนึ่ง  (JOIN + WHERE)

  r.id,
  u.name AS requesterName,
  u.department,
  r.request_type,
  r.location,

FROM requests AS r
JOIN users AS u
  ON r.requester_id = u.id
WHERE u.department = 'วิศวกรรมซอฟต์แวร์'
